**1) 개인화 스코어 계산 함수**
```typescript
// src/lib/personalization/scorer.ts
import { prisma } from '@/lib/db/prisma';

export interface PersonalizationScore {
  placeId: string;
  personalScore: number; // 0-100
  popularityScore: number; // 0-100
  finalBoost: number; // -10 ~ +10
}

/**
 * 사용자의 과거 클릭 패턴 기반 개인화 점수 계산
 */
export async function calculatePersonalizationScores(
  sessionId: string,
  routeHash: string,
  placeIds: string[]
): Promise<PersonalizationScore[]> {
  // 1. 개인 클릭 히스토리 조회 (최근 30일)
  const thirtyDaysAgo = new Date(Date.now() - 30 * 24 * 60 * 60 * 1000);
  
  const userClicks = await prisma.clickLog.groupBy({
    by: ['placeId'],
    where: {
      sessionId,
      clickedAt: { gte: thirtyDaysAgo },
      placeId: { in: placeIds },
    },
    _count: { placeId: true },
  });

  // 2. 경로별 인기도 조회 (같은 경로에서 다른 사용자들의 선택)
  const routePopularity = await prisma.clickLog.groupBy({
    by: ['placeId'],
    where: {
      searchLog: {
        routeHash, // 출발+도착 해시
      },
      placeId: { in: placeIds },
    },
    _count: { placeId: true },
  });

  // 3. 점수 계산
  const maxUserClicks = Math.max(...userClicks.map(c => c._count.placeId), 1);
  const maxRouteClicks = Math.max(...routePopularity.map(c => c._count.placeId), 1);

  return placeIds.map(placeId => {
    const userClickCount = userClicks.find(c => c.placeId === placeId)?._count.placeId || 0;
    const routeClickCount = routePopularity.find(c => c.placeId === placeId)?._count.placeId || 0;

    const personalScore = (userClickCount / maxUserClicks) * 100;
    const popularityScore = (routeClickCount / maxRouteClicks) * 100;

    // 최종 boost: 개인 70%, 인기 30%
    const finalBoost = (personalScore * 0.7 + popularityScore * 0.3) / 10 - 5;
    // -5 ~ +5 범위 (평균 0)

    return {
      placeId,
      personalScore,
      popularityScore,
      finalBoost,
    };
  });
}
```

```typescript
// src/lib/cache/search-cache.ts
interface CachedSearchResult {
  data: any; // SearchWaypointsResponse
  timestamp: number;
  routeHash: string;
  category: string;
}

const CACHE_TTL_MS = 30 * 60 * 1000; // 30분
const CACHE_KEY_PREFIX = 'midwayder_search_';

export function getCachedSearch(
  routeHash: string,
  category: string
): CachedSearchResult | null {
  try {
    const key = `${CACHE_KEY_PREFIX}${routeHash}_${category}`;
    const cached = localStorage.getItem(key);
    
    if (!cached) return null;
    
    const parsed: CachedSearchResult = JSON.parse(cached);
    const age = Date.now() - parsed.timestamp;
    
    if (age > CACHE_TTL_MS) {
      // 만료된 캐시 삭제
      localStorage.removeItem(key);
      return null;
    }
    
    return parsed;
  } catch (error) {
    console.error('Cache read error:', error);
    return null;
  }
}

  routeHash: string,

    localStorage.setItem(key, JSON.stringify(cached));
    
    // 캐시 크기 제한: 최대 20개 유지
    cleanupOldCaches();
  } catch (error) {
    console.error('Cache write error:', error);
  }
}

function cleanupOldCaches(): void {
  const keys = Object.keys(localStorage).filter(k => k.startsWith(CACHE_KEY_PREFIX));
  
  if (keys.length <= 20) return;
  
  // timestamp 기준 정렬 후 오래된 것 삭제
  const caches = keys.map(key => {
    const data = JSON.parse(localStorage.getItem(key)!);
    return { key, timestamp: data.timestamp };
  });
  
  caches.sort((a, b) => a.timestamp - b.timestamp);
  
  // 가장 오래된 5개 삭제
  caches.slice(0, 5).forEach(cache => {
    localStorage.removeItem(cache.key);
  });
}
```

```typescript
        set({

          originalRoute: cached.data.originalRoute,
          totalCandidates: cached.data.totalCandidates,
          apiCallsUsed: cached.data.apiCallsUsed,
          isLoading: false,
          hasSearched: true,
        });
        return;
      }
      
      console.log('❌ Cache MISS:', routeHash, category);
      
      // 2. API 호출
      const response = await fetch('/api/search', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ start, end, category, options }),
      });

      if (!response.ok) {
        throw new Error('검색 실패');
      }

      const data = await response.json();

      set({
        results: data.results,
```
