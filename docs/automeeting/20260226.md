#### 시나리오 1: 첫 방문 사용자
```
1. 앱 접속 → "이게 뭐하는 앱이지?" (설명 부족)
2. 검색 시도 → 결과 나옴

- 앱의 가치가 1회성 경험으로 끝남

- 온보딩에서 "실제 시나리오" 보여주기 (예: "퇴근길 GS25 들러서 우유 사기")

// lib/routine/detector.ts
export type RoutineType = 'morning-commute' | 'evening-commute' | 'weekend-trip' | null;

export function detectRoutine(): RoutineType {
  const now = new Date();
  const hour = now.getHours();
  const day = now.getDay();
  
  // 평일 (월~금)
  if (day >= 1 && day <= 5) {
    if (hour >= 7 && hour <= 9) return 'morning-commute';
    if (hour >= 17 && hour <= 19) return 'evening-commute';
  }
  
  // 주말 (토~일)
  if (day === 0 || day === 6) {
    if (hour >= 10 && hour <= 18) return 'weekend-trip';
  }
  
  return null;
}

export function getRoutineLabel(type: RoutineType): string {
  const labels: Record<NonNullable<RoutineType>, string> = {
    'morning-commute': '출근 경로',
    'evening-commute': '퇴근 경로',
    'weekend-trip': '주말 나들이',
  };
  return type ? labels[type] : '';
}

export function shouldShowRoutinePrompt(type: RoutineType, lastPromptTime: number | null): boolean {
  if (!type) return false;
  if (!lastPromptTime) return true;
  
  // 같은 루틴 타입에 대해 4시간에 한 번만 프롬프트
  const fourHours = 4 * 60 * 60 * 1000;
  return Date.now() - lastPromptTime > fourHours;
}
```

// app/page.tsx
import { detectRoutine, getRoutineLabel, shouldShowRoutinePrompt } from '@/lib/routine/detector';

export default function HomePage() {
  // ... 기존 state들 ...
  const [routinePromptDismissed, setRoutinePromptDismissed] = useState(false);

  // 루틴 감지 (앱 로드 시)
  useEffect(() => {
    const routine = detectRoutine();
    if (!routine) return;

    const lastPromptKey = `routine-prompt-${routine}`;
    const lastPromptTime = localStorage.getItem(lastPromptKey);
    const lastTime = lastPromptTime ? parseInt(lastPromptTime, 10) : null;

    if (shouldShowRoutinePrompt(routine, lastTime) && !routinePromptDismissed) {
      // 즐겨찾기 중 해당 루틴 찾기
      const routineFavorite = favorites.find(f => f.routineType === routine);
      
      if (routineFavorite) {
        // 프롬프트 표시
        const label = getRoutineLabel(routine);
        const confirmLoad = window.confirm(
          `${label}을(를) 불러올까요?\n${routineFavorite.name}`
        );
        
        if (confirmLoad) {
          setStart(routineFavorite.start);
          setEnd(routineFavorite.end);
          setCategory(routineFavorite.category);
          setSearchOverlayOpen(true);

#### API 엔드포인트
```typescript
// app/api/verify-visit/route.ts
import { NextRequest, NextResponse } from 'next/server';
import { prisma } from '@/lib/db/prisma';

function calculateDistance(lat1: number, lng1: number, lat2: number, lng2: number): number {
  const R = 6371000; // 지구 반지름 (미터)
  const dLat = ((lat2 - lat1) * Math.PI) / 180;
  const dLng = ((lng2 - lng1) * Math.PI) / 180;
  const a =
    Math.sin(dLat / 2) * Math.sin(dLat / 2) +
    Math.cos((lat1 * Math.PI) / 180) *
      Math.cos((lat2 * Math.PI) / 180) *
      Math.sin(dLng / 2) *
      Math.sin(dLng / 2);
  const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
  return R * c;
}

export async function POST(request: NextRequest) {
  try {
    const { placeId, userLat, userLng } = await request.json();
    const sessionId = request.cookies.get('sessionId')?.value || 'anonymous';

    // 장소 정보 조회
    const place = await prisma.place.findUnique({
      where: { id: placeId },
    });

    if (!place) {
      return NextResponse.json({ error: 'Place not found' }, { status: 404 });
    }

    // 거리 계산 (50m 이내면 인증)
    const distance = calculateDistance(userLat, userLng, place.lat, place.lng);

    if (distance > 50) {
      return NextResponse.json({
        verified: false,
        message: `장소에서 ${Math.round(distance)}m 떨어져 있어요`,
      });
    }

    // 중복 인증 방지 (1시간 이내)
    const recentVisit = await prisma.visitLog.findFirst({
      where: {
        placeId,
        sessionId,
        verifiedAt: {
          gte: new Date(Date.now() - 60 * 60 * 1000), // 1시간
        },
      },
    });

    if (recentVisit) {
      return NextResponse.json({
        verified: false,
        message: '이미 인증한 장소예요',
      });
    }

    // 방문 로그 저장
    const points = 10;
    await prisma.visitLog.create({
      data: {
        placeId,
        sessionId,
        points,
      },
    });

    // 포인트 적립
    const userPoints = await prisma.userPoints.upsert({
      where: { sessionId },
      create: { sessionId, points },
      update: { points: { increment: points } },
    });

    // 티어 업데이트
    let tier = 'bronze';
    if (userPoints.points >= 500) tier = 'platinum';
    else if (userPoints.points >= 200) tier = 'gold';
    else if (userPoints.points >= 50) tier = 'silver';

    if (tier !== userPoints.tier) {
      await prisma.userPoints.update({
        where: { sessionId },
        data: { tier },
      });
    }

    return NextResponse.json({
      verified: true,
      points,
      totalPoints: userPoints.points + points,
      tier,
      message: `${points} 포인트 적립! 총 ${userPoints.points + points}점`,
    });
  } catch (error: any) {
    console.error('[VerifyVisit] Error:', error);
    return NextResponse.json({ error: error.message }, { status: 500 });
  }
}
```

// components/place/PlaceDetail.tsx (방문 인증 버튼 추가)
export default function PlaceDetail({ place, onClose }: PlaceDetailProps) {
  const [verifying, setVerifying] = useState(false);
  const [verifyResult, setVerifyResult] = useState<{ verified: boolean; message: string; points?: number } | null>(null);

  const handleVerifyVisit = async () => {
    setVerifying(true);

    try {
      // GPS 권한 요청
      const position = await new Promise<GeolocationPosition>((resolve, reject) => {
        navigator.geolocation.getCurrentPosition(resolve, reject, {
          enableHighAccuracy: true,
          timeout: 10000,
        });
      });

      const { latitude, longitude } = position.coords;

      // 인증 API 호출
      const response = await fetch('/api/verify-visit', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          placeId: place.id,
          userLat: latitude,
          userLng: longitude,
        }),
      });

      const data = await response.json();
      setVerifyResult(data);

      if (data.verified) {
        showToast(`🎉 ${data.message}`, 'success');
      } else {
        showToast(data.message, 'error');
      }
    } catch (error: any) {
      showToast('GPS 권한이 필요해요', 'error');
    } finally {
      setVerifying(false);
    }
  };

  return (
    <BottomSheet open={true} onClose={onClose} snap="half">
      <div className="p-6">
        <h3 className="text-xl font-bold">{place.name}</h3>
        <p className="text-sm text-gray-600">{place.address}</p>

        {/* 방문 인증 버튼 */}
        <button
          onClick={handleVerifyVisit}
          disabled={verifying}
          className="w-full mt-4 py-3 bg-green-500 text-white rounded-lg font-medium disabled:bg-gray-300"
        >
          {verifying ? '인증 중...' : '🎯 방문 인증하고 포인트 받기'}
        </button>

        {verifyResult && (
          <div className={`mt-2 p-3 rounded-lg ${verifyResult.verified ? 'bg-green-100' : 'bg-red-100'}`}>
            <p className="text-sm">{verifyResult.message}</p>
          </div>
        )}
      </div>
    </BottomSheet>
  );
}
