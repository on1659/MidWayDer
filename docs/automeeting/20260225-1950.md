// lib/navigation-deeplinks.ts
export function generateKakaoNaviUrl(start, waypoint, end) {
  return `kakaomap://route?sp=${start.lat},${start.lng}&ep=${end.lat},${end.lng}&via=${waypoint.coordinates.lat},${waypoint.coordinates.lng}`;
}

export function generateTmapUrl(start, waypoint, end) {
  // 티맵 URL Scheme
  return `tmap://route?startx=${start.lng}&starty=${start.lat}&goalx=${end.lng}&goaly=${end.lat}&via1x=${waypoint.coordinates.lng}&via1y=${waypoint.coordinates.lat}`;
}

export function generateNaverMapUrl(start, waypoint, end) {
  // 네이버지도 URL Scheme
  return `nmap://route/car?slat=${start.lat}&slng=${start.lng}&sname=출발지&dlat=${end.lat}&dlng=${end.lng}&dname=도착지&v1lat=${waypoint.coordinates.lat}&v1lng=${waypoint.coordinates.lng}&v1name=${encodeURIComponent(waypoint.name)}`;
}

export function generateAppleMapsUrl(start, waypoint, end) {
  return `maps://?saddr=${start.lat},${start.lng}&daddr=${end.lat},${end.lng}`;
}
```

// POST /api/reviews
export async function POST(request: NextRequest) {
  const { placeId, rating, comment, tags } = await request.json();
  const sessionId = request.cookies.get('sessionId')?.value;

  const review = await prisma.visitFeedback.create({
    data: {
      placeId,
      rating,
      comment,
      tags: tags.join(','),
      sessionId,
    }
  });

  return NextResponse.json({ success: true, review });
}

// GET /api/reviews?placeId=xxx
export async function GET(request: NextRequest) {
  const placeId = request.nextUrl.searchParams.get('placeId');
  
  const [reviews, stats] = await Promise.all([
    prisma.visitFeedback.findMany({
      where: { placeId },
      orderBy: { timestamp: 'desc' },
      take: 50
    }),
    prisma.visitFeedback.aggregate({
      where: { placeId },
      _avg: { rating: true },
      _count: { id: true }
    })
  ]);

  return NextResponse.json({
    success: true,
    reviews,
    averageRating: stats._avg.rating || 0,
    reviewCount: stats._count.id
  });
}
```

// lib/cost-calculator.ts
export function calculateFuelCost(distanceM: number): number {
  const distanceKm = distanceM / 1000;
  const fuelUsed = distanceKm / 10; // 연비 10km/L
  const fuelPrice = 1600; // 휘발유 가격 (원/L)
  return Math.round(fuelUsed * fuelPrice);
}

export function calculateTimeCost(durationSec: number): number {
  const durationHour = durationSec / 3600;
  const hourlyWage = 15000; // 시급
  return Math.round(durationHour * hourlyWage);
}

```typescript
import type { Location } from '@/types/location';

interface NaviParams {
  start: Location;
  waypoint: { name: string; coordinates: { lat: number; lng: number } };
  end: Location;
}

/**
 * 카카오내비 딥링크
 */
export function generateKakaoNaviUrl({ start, waypoint, end }: NaviParams): string {
  const startCoords = start.coordinates;
  const endCoords = end.coordinates;
  const wpCoords = waypoint.coordinates;

  if (!startCoords || !endCoords) return '';

  // kakaomap://route?sp=lat,lng&ep=lat,lng&via=lat,lng
  return `kakaomap://route?sp=${startCoords.lat},${startCoords.lng}&ep=${endCoords.lat},${endCoords.lng}&via=${wpCoords.lat},${wpCoords.lng}`;
}

/**
 * 티맵 딥링크
 */
export function generateTmapUrl({ start, waypoint, end }: NaviParams): string {
  const startCoords = start.coordinates;
  const endCoords = end.coordinates;
  const wpCoords = waypoint.coordinates;

  if (!startCoords || !endCoords) return '';

  // tmap://route?startx=lng&starty=lat&goalx=lng&goaly=lat&via1x=lng&via1y=lat
  return `tmap://route?startx=${startCoords.lng}&starty=${startCoords.lat}&goalx=${endCoords.lng}&goaly=${endCoords.lat}&via1x=${wpCoords.lng}&via1y=${wpCoords.lat}`;
}

/**
 * 네이버지도 딥링크
 */
export function generateNaverMapUrl({ start, waypoint, end }: NaviParams): string {
  const startCoords = start.coordinates;
  const endCoords = end.coordinates;
  const wpCoords = waypoint.coordinates;

  if (!startCoords || !endCoords) return '';

  // nmap://route/car?slat=&slng=&sname=&dlat=&dlng=&dname=&v1lat=&v1lng=&v1name=
  return `nmap://route/car?slat=${startCoords.lat}&slng=${startCoords.lng}&sname=${encodeURIComponent(start.address || '출발지')}&dlat=${endCoords.lat}&dlng=${endCoords.lng}&dname=${encodeURIComponent(end.address || '도착지')}&v1lat=${wpCoords.lat}&v1lng=${wpCoords.lng}&v1name=${encodeURIComponent(waypoint.name)}`;
}

/**
 * Apple Maps 딥링크 (경유지 미지원)
 */
export function generateAppleMapsUrl({ start, end }: NaviParams): string {
  const startCoords = start.coordinates;
  const endCoords = end.coordinates;

  if (!startCoords || !endCoords) return '';

  return `maps://?saddr=${startCoords.lat},${startCoords.lng}&daddr=${endCoords.lat},${endCoords.lng}`;
}

/**
 * iOS 감지
 */
export function isIOS(): boolean {
  if (typeof window === 'undefined') return false;
  return /iPad|iPhone|iPod/.test(navigator.userAgent);
}
```

#### 1. **인기 경유지 배지 시스템 (25분)**
**무엇을 만들었나:**
- 최근 30일 클릭 로그 기반 인기도 점수 계산
- 상위 30% 경유지에 "🏆 인기" 배지 표시
- PlaceDetail에 "X명이 선택했어요" 상세 정보

**왜 중요한가:**
- 사회적 증거 → 의사결정 속도 향상
- 수집한 데이터를 UX 개선에 활용
- 신규 사용자 신뢰도 증가

**기술 포인트:**
- Prisma groupBy로 클릭 집계
- 정규화 (0-100) 점수 계산
- 비동기 로깅으로 성능 영향 제로

---

#### 2. **네비게이션 바로가기 강화 (30분)**
**무엇을 만들었나:**
- 4개 네비 앱 지원:
  - 🗺️ 카카오내비 (기존)
  - 📍 티맵 (신규, 국내 1위)
  - 🧭 네이버지도 (신규)
  - 🍎 Apple Maps (iOS만, 신규)
- 경유지 좌표 자동 포함
- 앱별 색상/아이콘 통일

**왜 중요한가:**
- 사용자 선택권 확대 (네비 앱 다양성)
- 이탈률 감소 (바로 네비 시작 가능)
- 크로스플랫폼 지원 (iOS/Android)

**기술 포인트:**
- URL Scheme 딥링크 (각 앱별 스펙)
- iOS 감지 (Apple Maps 조건부 표시)
- 경유지 via 파라미터 활용

---

#### 3. **추천 이유 명시 (15분)**
**무엇을 만들었나:**
- PlaceDetail에 "왜 추천했나요?" 섹션
  - ✅ 이탈 거리 (X km만 벗어남)
  - ✅ 이탈 시간 (X분만 더 걸림)
  - ✅ 인기도 (X명이 선택)
  - ✅ 경로 근접도 (높은 점수)
- ResultList 카드에 배지 추가
  - 🎯 이탈 최소 (<500m)
  - 🏆 인기 (상위 30%)
  - 📍 경로상 (근접도 90+)
  - ⚡ 빠름 (<3분)

**왜 중요한가:**
- 알고리즘 투명성 → 신뢰도 향상
- 교육 효과 (사용자가 판단 기준 학습)
- 클릭률 증가 (납득 가능한 이유)

**기술 포인트:**
- detourCost, proximityScore, clickCount 조합
- 조건부 렌더링 (데이터 있을 때만 표시)
- 시각적 일관성 (Check 아이콘, 파스텔 배경)
