### A. 🍞 **Toast 알림 시스템 - "alert() 퇴출" 📢**

**현재 상태:**
```tsx
// 7곳에서 alert() 사용 중
alert('링크가 복사되었습니다!');
alert('GPS 권한이 필요해요');
alert('이 브라우저에서는 위치 기능을 사용할 수 없어요');
```

**문제:**
- alert()는 2000년대 느낌
- 브라우저 네이티브 팝업으로 사용자 경험 단절
- 모바일에서 특히 불편 (화면 중앙 차단)
- 비동기 작업 블로킹

**개선안:**
```tsx
// useToast hook
export function useToast() {
  const [toasts, setToasts] = useState<Toast[]>([]);

  const showToast = (message: string, type: 'success' | 'error' | 'info' = 'info') => {
    const id = Date.now();
    setToasts(prev => [...prev, { id, message, type }]);
    setTimeout(() => {
      setToasts(prev => prev.filter(t => t.id !== id));
    }, 3000);
  };

  return { toasts, showToast };
}

// Toast Container
<div className="fixed top-4 right-4 z-50 space-y-2 pointer-events-none">
  {toasts.map(toast => (
    <div
      key={toast.id}
      className={`px-4 py-3 rounded-xl shadow-lg animate-slide-in pointer-events-auto ${
        toast.type === 'success' ? 'bg-green-500 text-white' :
        toast.type === 'error' ? 'bg-red-500 text-white' :
        'bg-blue-500 text-white'
      }`}
    >
      <div className="flex items-center gap-2">
        {toast.type === 'success' && '✅'}
        {toast.type === 'error' && '⚠️'}
        {toast.type === 'info' && 'ℹ️'}
        <span className="font-medium">{toast.message}</span>
      </div>
    </div>
  ))}
</div>

// 사용 예시
showToast('주소를 복사했어요! 📋', 'success');
showToast('GPS 권한이 필요해요', 'error');
showToast('검색 결과를 찾았어요', 'info');
```

**CSS 애니메이션:**
```css

  }
  to {
    opacity: 1;
    transform: translateX(0);
  }
}

  animation: slide-in 0.3s ease-out;
}
```

#### 파일 생성: `src/hooks/useToast.ts`

```typescript
import { useState } from 'react';

export interface Toast {
  id: number;
  message: string;
  type: 'success' | 'error' | 'info';
}

export function useToast() {
  const [toasts, setToasts] = useState<Toast[]>([]);

  const showToast = (message: string, type: 'success' | 'error' | 'info' = 'info') => {
    const id = Date.now();
    const newToast: Toast = { id, message, type };
    
    setToasts(prev => [...prev, newToast]);
    
    setTimeout(() => {
      setToasts(prev => prev.filter(t => t.id !== id));
    }, 3000);
  };

  return { toasts, showToast };
}
```

```tsx
          mapRef.current.setCenter(new kakao.maps.LatLng(latitude, longitude));
          mapRef.current.setLevel(3);
        }
      },
      (error) => {
        alert('위치를 가져올 수 없어요');
      }
    );
  };

  return (
    <div className="relative w-full h-full">
      {/* 기존 지도 */}
      <div ref={mapContainerRef} className="w-full h-full" />
      
      {/* 지도 컨트롤 버튼 */}
      <div className="absolute bottom-24 right-4 flex flex-col gap-2 z-10">
        <button
          onClick={handleZoomIn}
          className="w-10 h-10 bg-white rounded-full shadow-lg flex items-center justify-center hover:bg-gray-50 active:scale-95 transition-all"
          aria-label="확대"
        >
          <Plus className="w-5 h-5" style={{ color: 'var(--text-primary)' }} />
        </button>
        
        <button
          onClick={handleZoomOut}
          className="w-10 h-10 bg-white rounded-full shadow-lg flex items-center justify-center hover:bg-gray-50 active:scale-95 transition-all"
          aria-label="축소"
        >
          <Minus className="w-5 h-5" style={{ color: 'var(--text-primary)' }} />
        </button>
        
        <div className="h-px bg-gray-200 my-1" />
        
        <button
          onClick={handleGPSToMap}
          className="w-10 h-10 bg-white rounded-full shadow-lg flex items-center justify-center hover:bg-gray-50 active:scale-95 transition-all"
          aria-label="내 위치"
        >
          <LocateFixed className="w-5 h-5" style={{ color: 'var(--accent)' }} />
        </button>
      </div>
    </div>
  );
}
```

```typescript
export interface RecentSearch {
  startAddress: string;
  endAddress: string;
  startCoords?: { lat: number; lng: number };
  endCoords?: { lat: number; lng: number };
  category: string;
  timestamp: number;
}

export function addRecentSearch(search: Omit<RecentSearch, 'timestamp'>) {
  try {
    const searches = getRecentSearches();
    const newSearch: RecentSearch = { ...search, timestamp: Date.now() };
    
    // 중복 제거 (같은 출발/도착/카테고리)
    const filtered = searches.filter(
      s => !(s.startAddress === search.startAddress && 
             s.endAddress === search.endAddress && 
             s.category === search.category)
    );
    
    filtered.unshift(newSearch);
    localStorage.setItem('recentSearches', JSON.stringify(filtered.slice(0, 10)));
  } catch { /* ignore */ }
}
```
